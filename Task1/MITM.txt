--Running Client Container
docker run -it --name client --network host ubuntu

--Updating and Installing swaks on Client Container
apt update
apt install -y swaks

--sending mail using swaks in client container
swaks --to bob@example.com \
  --from alice@example.com \
  --server 127.0.0.1:1025 \
  --header "Subject: MITM Test" \
  --body "Hi Bob,
  This is a plaintext email.
  Regards,
  Alice"

--Running Attacker Client
docker run -it --name attacker --network host ubuntu

--Updating and installing -y tcpdump
apt update
apt install -y tcpdump

--Starting tcpdump
tcpdump -A -i any port 1025
